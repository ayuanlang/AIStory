        elif use_runway_api:

        elif is_kling_3_video:
            kling_model = "kling-3.0/video"

            def _normalize_bool(raw: Any, default: bool) -> bool:
                if raw is None:
                    return default
                if isinstance(raw, bool):
                    return raw
                text = str(raw).strip().lower()
                if text in {"1", "true", "yes", "y", "on"}:
                    return True
                if text in {"0", "false", "no", "n", "off"}:
